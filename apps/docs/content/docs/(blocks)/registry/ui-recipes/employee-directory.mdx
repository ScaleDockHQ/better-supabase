It loads its data on the server through the block's client; the [employees block](https://bettersupabase.com/blocks/employees) must be installed (`npx better-supabase add employees`). Its page on bettersupabase.com is [/recipes/employee-directory](https://bettersupabase.com/recipes/employee-directory).

Web, with no shadcn components.

```tsx title="apps/examples/nextjs/src/features/people/components/employee-overview.tsx"
/**
 * Render inside `<Suspense>`: every employee on the list kit, with the add
 * action for callers who can write.
 */
export async function EmployeeOverview() {
  const [t, permissions] = await Promise.all([
    getExtracted("people"),
    getPermissions(),
  ]);

  if (!can(permissions, "employees.read")) {
    forbidden();
  }

  const directory = await settle(loadDirectory);

  if (directory.kind === "no-organization") {
    return null;
  }

  if (directory.kind === "error") {
    return <PeopleLoadError title={t("The directory didn't load")} />;
  }

  const managers = directory.data.map((employee) => ({
    value: employee.id,
    label: employee.name,
  }));

  return (
    <div className="flex flex-col gap-4">
      {can(permissions, "employees.write") ? (
        <div className="flex justify-end">
          <AddEmployeeDialog
            managers={managers}
            canViewSensitive={can(permissions, "employees.read_sensitive")}
          />
        </div>
      ) : null}
      <EmployeeRecordList
        initial={pageBlockRows(
          employeeList,
          employeeSpec,
          directory.data,
          employeeList.defaults,
        )}
      />
    </div>
  );
}
```
